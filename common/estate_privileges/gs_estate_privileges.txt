estate_genshin_base = {
	icon = privilege_grant_autonomy

	land_share = 28 				#要让出的王冠领地（给阶级）

	#is_valid = { always = yes }	#国家启用该特权的条件
	can_select = {
		if = {
			limit = { gs_has_OnlyFans_monuments = yes }
			OR = {
				gs_has_OnlyFans_leader = yes
				gs_has_gs_leader_with_element = yes
				has_country_modifier = gs_Paimon
			}
		}
		else = {
			OR = {
				gs_has_gs_leader_with_element = yes
				has_country_modifier = gs_Paimon
			}
		}
	}		#国家是否允许选择特权条件
	#on_granted = { <effects> }     #特权立即执行的效果
	can_revoke = {
		NOT = { has_estate_privilege = estate_genshin_leader }
		NOT = { has_estate_privilege = estate_genshin_peiou }
		NOT = { has_estate_privilege = estate_genshin_advisor }
	} 	#国家是否可以撤销特权的条件 
	#on_revoked = { <effects> } 	#特权被撤销的影响
	#on_invalid = { <effects> } 	#由于权限无效造成撤销的影响

	#penalties = { <modifiers> } 	#特权激活时代负面修正
	benefits = {
		governing_capacity = 300
		country_admin_power = 1
		country_diplomatic_power = 1
		country_military_power = 1
		global_tax_modifier = 0.1
		trade_efficiency = 0.1
		global_manpower_modifier = 0.1
		estate_genshin_privilege_slots = 49
	}      #特权激活时的正面修正
	#conditional_modifier = { 		
	#	#如果指令为真，应用的修正
	#		 trigger = { <triggers> }
	#		 modifier = { <modifiers> }
	#		 is_bad = <bool>				#yes/no修正是否是负面的
	#}
	#modifier_by_land_ownership = { <modifiers> } 	#修正与阶级领地相乘
	#mechanics = { <mechanics> } 			#特权引用的特殊机制

	#cooldown_years = <int> 				#特权无法撤销的时间限制
	#on_cooldown_expires = { effects }		#Effect executed once the specified cooldown_years timer has run down

	ai_will_do = {					#If above ESTATE_PRIVILEGE_GRANT_THRESHOLD define, AI will try to apply. If below 
		factor = 1000                               #ESTATE_PRIVILEGE_REVOKE_THRESHOLD, AI will try to revoke.
		#modifier = {
		#		factor = <float>
		#		<triggers>
		#}
