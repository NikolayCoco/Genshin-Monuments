#国家
on_startup = {
	hidden_effect = {
		if = {
			limit = { NOT = { has_global_flag = gs_start_end } }
			set_global_flag = ys_qiguan_shengfen_mingzi
			set_global_flag = gs_allow_ai_Easter
			set_global_flag = gs_allow_ai_gs_estate
			set_global_flag = gs_monuments_amplitude_3_player
			set_global_flag = gs_monuments_amplitude_3_ai
			set_global_flag = gs_start_end
			if = {
				limit = { has_global_flag = gs_move_start }
				set_global_flag = ys_ai_qiguan_shengfen_mingzi
			}
		}
	}
	if = {
		limit = {
			ai = no
			NOT = { has_country_modifier = gs_start }
		}
		country_event = { id = gs_decisions_events.1000 }
	}
}
on_bankruptcy = {
	if = {
		limit = {
			has_country_modifier = gs_Paimon
			NOT = { has_country_flag = gs_Paimon_repay_end }
		}
		set_country_flag = gs_Paimon_repay_end
		country_event = { id = gs_decisions_events.1003 }
	}
}
on_new_monarch = {
	hidden_effect = {
		#宝玦口防水灾
		if = {
			limit = {
				OR = {
					has_country_modifier = gsld_gov_reform_gs_S_12
					has_country_modifier = gs_baojue_fanghong
				}
			}
			country_event = { id = gs_monuments_events.15 }
		}
	}
}

					religion = yangmingism
					#religion = elysia_ruru_worship

	}
