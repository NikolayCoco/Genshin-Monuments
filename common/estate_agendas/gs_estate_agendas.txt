#(Nearly) all scopes are country scope
#All parameters are optional (but it will not be much of an agenda without some of them...)

estate_genshin_build_monument = {
	#max_days_active = <float> 		#Lets you override the default of 20 years for an agenda
	can_select = {
		any_owned_province = {
			has_great_project = {
				type = any
				tier = 0
			}
			NOT = {
				has_great_project = {
					type = any
					tier = 3
				}
			}
		}
	} 		#Conditions for whether the agenda can be offered to you
	selection_weight = {			#Determines how likely (compared to the estate's other agendas) this agenda is to be selected.
		factor = 1		#initial factor will ignore numbers below 1, use a modifier for that!
	#	modifier = { factor = <float> triggers = <triggers> }
	}

	pre_effect = {
		random_owned_province = {
			limit = {
				has_great_project = {
					type = any
					tier = 0
				}
				NOT = {
					has_great_project = {
						type = any
						tier = 3
					}
				}
			}
			save_event_target_as = agenda_province
		}
	}		#Effects executed to set up the task (set flags and targets), before it is shown to the player.
	provinces_to_highlight = { province_id = event_target:agenda_province }	#Conditions for provinces to be highlighted by the agenda. Root = country, this = province.
	#immediate_effect = { effects } 		#Effects executed when the agenda is accepted by the country
	task_requirements = {
		event_target:agenda_province = {
			has_great_project = {
				type = any
				tier = 3
			}
		}
	} 	#Conditions for completing the task successfully
	task_completed_effect = {
		on_completed_agenda_effect = {
			estate = estate_genshin
		}
		add_estate_loyalty = {
			estate = estate_genshin
			loyalty = 10
		}
		add_prestige = 50
		add_adm_power = 100
		add_dip_power = 100
		add_dip_power = 100
	} 	#Effects executed when the agenda is successfully completed
	failing_effect = {
		on_failed_agenda_effect = yes
		clr_auto_complete_flag = yes
        add_estate_loyalty_modifier = {
            estate = estate_genshin
			desc = EST_VAL_AGENDA_DENIED
			loyalty = -10
			duration = 7300
        }
		add_prestige = -50
		add_adm_power = -100
		add_dip_power = -100
		add_dip_power = -100
	} 		#Effects executed when the agenda is failed (i.e. when fail_if is true or time runs out)
	fail_if = { event_target:agenda_province = { NOT = { owned_by = root } } } 			#Conditions for when the agenda should be auto-failed
	#invalid_trigger = { triggers } 		#Conditions for when the agenda should end without it counting as a fail (executing on_invalid 				 
											   #rather than failing_effect). Happens by default it the estate giving the agenda is removed.
	#on_invalid = { effects } 		#Effects executed when the country no longer has that estate or the invalid_trigger is true.
											   #Useful for clearing flags, modifiers etc.
	modifier = {
		great_project_upgrade_cost = -0.5
		great_project_upgrade_time = -0.5
	}
}
